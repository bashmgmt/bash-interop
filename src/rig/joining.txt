# The address of a session is its workspace directory. Loading its two laid
# files defines the words; initiation is the client's own line. A shell under
# a driven run with `--reach bash-env` needs neither — the provisioned
# startup file did both — and under `--reach by-hand` finds the workspace in
# BC_SESSION and joins where its script says so:
source "$BC_SESSION/prelude.bash"
source "$BC_SESSION/rig.bash"
BASHPROF_INIT "$BC_SESSION"

# A script that starts the tool itself, as a coprocess: it names and makes
# the workspace, probes the same directory until the session is up, loads,
# initiates, and holds the session open for as long as it runs:
source lib/joining.bash
declare -- workspace="$PWD/prof.d"             # an address is absolute
mkdir -p "$workspace"
BC_START bashprof serve --at "$workspace" --into build.times
until BC_UP "$workspace"; do sleep 0.01; done
BC_LOAD "$workspace"
BASHPROF_INIT "$workspace"
BC_LEAVE                                       # let go, wait, return its status

# Only if there is a session — a script that runs the same with or without one:
if [[ -n ${BC_SESSION-} ]]; then
    BC_LOAD "$BC_SESSION"
    BASHPROF_INIT "$BC_SESSION"
fi

# The words a client vendors, and the polyfill for when nothing is listening —
# after the load, so a session's own hooks win:
source lib/bashprof.bash
declare -F __bp_begin >/dev/null || { __bp_begin() { :; }; __bp_end() { :; }; }
