# The address of a session is the file a shell sources to join it. A shell
# under a driven run finds it in its environment as BC_SESSION; under
# --reach bash-env every shell has already sourced it, and otherwise a shell
# joins where its script says so:
source "$BC_SESSION"

# A script that starts the tool itself, as a coprocess, holds the session
# open for as long as it runs and joins as above:
source lib/joining.bash
BC_START bashprof serve --into build.times     # start, take the address, join
BC_LEAVE                                       # let go, wait, return its status

# Only if there is a session — a script that runs the same with or without one:
[[ -n ${BC_SESSION-} ]] && source "$BC_SESSION"

# The words a client vendors, and the polyfill for when nothing is listening —
# after the join, so a session's own hooks win:
source lib/bashprof.bash
declare -F __bp_begin >/dev/null || { __bp_begin() { :; }; __bp_end() { :; }; }
